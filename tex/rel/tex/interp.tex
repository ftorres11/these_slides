    \begin{columns}

            \begin{center}
                {\scriptsize Input image. ($I$)}\\
                \includegraphics[width=.5\textwidth]{fig/rel/interp/img/ILSVRC2012_val_00000027.JPEG}\\
                {\scriptsize $P^c_i = 0.8756$}\\
            \end{center}
        %\column{0.3\textwidth}

                {\scriptsize Explanation Map. ($E^c$)}\\
                \includegraphics[width=.5\textwidth]{fig/rel/interp/img/masked_resized.png}\\
                {\scriptsize $O^c_i = 0.7442$}
            \end{center}
        \pause
        \column[T]{.6\textwidth}
            {\scriptsize%
                \begin{itemize}
                    \item Prediction loss when using an explanation. (Average Drop 
                        (AD) \cite{chattopadhay2018grad})
                    \item Ratio of explanations being better than the input image. (Average Increase 
                        (AI) \cite{chattopadhay2018grad})
                    \item Reaction to saliency guided modifications (Insertion/Deletion 
                        (Ins/Del) \cite{petsiuk2018rise}) 
                \end{itemize}
            }
    \end{columns}

    \vspace{-10pt}
    \begin{columns}
        \column[T]{0.4\textwidth}
            \begin{center}
                {\scriptsize Input image. ($I$)}\\
                \includegraphics[width=.5\textwidth]{fig/rel/interp/img/ILSVRC2012_val_00000027.JPEG}\\
            \end{center}
        %\column{0.3\textwidth}
            \begin{center}
                {\scriptsize Saliency Map. ($E^c$)}\\
                \includegraphics[width=.5\textwidth]{fig/rel/interp/img/saliency.png}\\
            \end{center}
        \pause
        \column[T]{.6\textwidth}
            {\scriptsize%
                \begin{itemize}
                    \item Relationships between the bounding boxes (OM, LE, F$_1$).
                    \item Overlap between object mask and bounding box (BA).
                    \item Localization of most salient points and fraction of them in the bounding box (EP, SP).
                    \item Penalty of the size of the bounding box and image size, regarding prediction (SM).                        
                \end{itemize}
            }
    \end{columns}

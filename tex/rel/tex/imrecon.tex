\begin{frame}[t]
    \frametitle{Image Recognition Models}
    \framesubtitle{Self Attention Architectures}
    \begin{columns}
        \column[T]{.5\textwidth}
         {\small \textbf{Transformers}\\\vspace{10pt}
         \emph{Attention:} compute the relevance of each element relative to others in an 
         embedding.\\\vspace{10pt}
         {\scriptsize
            \begin{equation}
                \mbox{Attention}(Q, K, V) = \mbox{softmax}\left(\frac{QK^T}{\sqrt{d_k}}\right) V
                \label{eq:att}
            \end{equation}
         }
         \pause
         An abstract representation of a class is computed, updated with information across the 
         embedding \cite{devlin2018bert}.
         }
        \column[T]{.5\textwidth}
        \begin{center}
            \includegraphics[width=.5\textwidth]{fig/rel/imrecon/img/scaled_dotproduct.pdf}

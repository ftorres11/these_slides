    \small
    Assuming target labels $T$:
    \begin{equation}
        L_R(X, \theta, T) = \frac{1}{n} \sum_{i=1}^n E(\delta x_i, \delta_G x_i),
    \end{equation}
    \pause
    And the total loss:
    \begin{equation}
        L(x, \theta, t) = L_C(x, \theta, t) + \lambda L_R(x, \theta, t),
    \label{eq:total}
    \end{equation}

\subsection*{Opti-CAM}
%--------------------------------------------------------------------------------------------------
\begin{frame}[t]
    \frametitle{Opti-CAM}
    \framesubtitle{Motivation}
    {\small 
        \begin{itemize}
            \item \textbf{Combination of Feature Maps}\\
            To compute $w^c_k$, Score-CAM \cite{wang2020score} considers feature maps
            independently.\\
            \emph{What about a linear combination?}\\
            \begin{equation*}
                F(\textbf{w})\defn f\left(\textbf{\mbox{u}}\odot n\left(\mbox{up}\left(\sum_k w_k 
                    A^k_l \right)\right)\right)_c
            \end{equation*}
            F is optimizable with respect to \textbf{w} and an initial \textbf{\mbox{$w_0$}} via 
            backpropagation.\pause
            \item \textbf{Masking-based Saliency}
            Saliency maps produced by optimization in the input space \cite{fong2017interpretable,
            fong2019understanding}\\
            The optimization usually takes form of:
            \begin{equation*}
                S^c(\mbox{\textbf{u}})\defn \underset{\mbox{m}\in M}{\mbox{argmax}}  
                      f\left(\mbox{\textbf{u}}\odot{n}\left(\mbox{up}(\textbf{\mbox{m}}) \right)\right)_c + \delta R(\mbox{\textbf{m}})
            \end{equation*}
        \end{itemize}
    }
\end{frame}

\begin{frame}[t]
    \frametitle{Opti-CAM}    
    \framesubtitle{Optimization}
    {\small
    \textbf{Formulation} Assuming channel weights $w_k\defn \mbox{softmax}(\textbf{u})_k$,
        the saliency map $S_\ell$ is a function of both an input image \textbf{$x$} and a 
        variable $\textbf{u}$:
        \begin{equation}
            S_\ell(\mbox{\textbf{x}}; \mbox{\textbf{u}})\defn \sum_k \mbox{softmax}(\textbf{\mbox{u}})_kA^k_\ell
        \end{equation}
    \vspace{-15pt}
    \pause
    \begin{columns}
        \column[T]{.6\textwidth}
        Find vector $\textbf{\mbox{u*}}$ maximizing confidence for class $c$, 
        with an input \textbf{x} masked by Saliency Map $S_\ell(\mbox{\textbf{x}}; \mbox{\textbf{u*}})$:
        \begin{equation}
            \mbox{\textbf{u*}}\defn \underset{\mbox{u}}{\mbox{argmax}}F^c_\ell (\mbox{\textbf{x}}; \mbox{\textbf{u*}}),
        \end{equation}
        \pause
        \column[T]{.4\textwidth}
        The objective function:
        \begin{equation}
            F^c_\ell(\textbf{\mbox{x}}; \textbf{\mbox{u}}) \defn g_c (f(\mbox{x}\odot)n)
        \end{equation} 
    \end{columns}
    \vspace{10pt}
    \pause
    The Saliency Map then becomes:
    \begin{equation}
        S^c_\ell(\textbf{x})\defn S_\ell(\textbf{\mbox{x}}; \textbf{\mbox{u*}}) =
        S_\ell (\textbf{\mbox{x}}; \underset{\mbox{u}}{\mbox{argmax}}F^c_\ell
        (\textbf{\mbox{x}}; \textbf{\mbox{u}}))
    \end{equation}    
    }
\end{frame}

\begin{frame}[t]
    \frametitle{Background}
    \framesubtitle{CAM Formalized}
    \small
    For an activation function ($h$) and  a feature map ($A^k_\ell$):
    \begin{equation}
        \small S^c_\ell (\textbf{\mbox{u}}) \defn h \left(\sum_k w^c_k A^k_\ell \right)
        \label{eq:cam}
    \end{equation}
    \pause
    The weighting coefficient ($\mathbf{w^c_k}$) defines the CAM variant.\\\pause
    Milestone CAM models:
    \vspace{10pt}
    \begin{columns}
        \scriptsize
        \column[T]{.33\textwidth}
            \textbf{CAM\cite{zhou2016learning}:}\\ $w^c_k$ comes from the weights for one class. 
            Only on the last convolutional layers followed by \Th{GAP} and simple classifiers. 
        \pause
        \column[T]{.33\textwidth}
            \textbf{Grad-CAM/Grad-CAM++ \cite{selvaraju2017grad, chattopadhay2018grad}:} \\$w^c_k$ is computed using gradients flowing from prediction to input space.\\
            It can be used at any point of the model. 
        \pause
        \column[T]{.33\textwidth}
            \textbf{Score-CAM\cite{wang2020score}:}\\$w^c_k$ is computed from the contributions of individual feature maps masking the input image.\\
        \pause
    \end{columns}
    \vspace{20pt}
    \small
    The approach found in Score-CAM is interesting. Diving deeper:
\end{frame}

\begin{frame}[t]
    \frametitle{Background}
    \framesubtitle{Score-CAM Formalized}
    \small
    Following equation (\ref{eq:cam}):\pause
    \begin{itemize}
        \item $h\defn\mathbf{ReLU}$
        \item $\mathbf{w^c_k} \defn softmax(\mathbf{u}^c)_k$
    \end{itemize}
    \pause
    \vspace{20pt}
    With feature map contribution ($\mathbf{u^c}_k$), model ($f$), normalization ($n$) and upsampling ($\mathbf{up}$):
    \begin{equation}
        u^c_k \defn f (\mbox{\textbf{x}} \odot n(\mbox{up}(A^k_\ell)))_c - f(\mbox{\textbf{x}})_c
        \label{eq:inc_conf}
    \end{equation}
    \vspace{10pt}
\end{frame}

\begin{frame}[t]
    \frametitle{Opti-CAM}
    \framesubtitle{Approach}
    \small
    \input{fig/opticam/common/tex/mainfig.tex}\pause
    Consider $S_\ell $, a function of $\mathbf{u}$ and $x$:
    \begin{equation}
        S_\ell(\mathbf{x; u})\defn \sum_k \mbox{softmax}(\mathbf{u})_kA^k_\ell
    \end{equation}
    \pause
    $\mathbf{u^*}$ is the vector that maximizes class $c$ probability, when masking the  the input with $S_\ell$.
    \begin{equation}
        \mathbf{u^*} \defn \underset{\mbox{u}}{\mbox{argmax}} F^c_\ell(\mathbf{x;u})
    \end{equation}
    And the objective $F^c_\ell(\mathbf{x;u}) \defn f(\mathbf{x}\odot n(\mbox{up}(S_\ell(\mathbf{x; u}))))$  

\end{frame}
